package yandex

import (
	"testing"
	"time"
)

func TestDualVolgaCarrierIndexRotatesWithoutGap(t *testing.T) {
	r := 25 * time.Second
	cases := []struct {
		elapsed time.Duration
		want    uint32
	}{
		{0, 0},
		{24*time.Second + 999*time.Millisecond, 0},
		{25 * time.Second, 1},
		{49*time.Second + 999*time.Millisecond, 1},
		{50 * time.Second, 0},
		{75 * time.Second, 1},
	}
	for _, tc := range cases {
		if got := dualVolgaCarrierIndex(tc.elapsed, r); got != tc.want {
			t.Fatalf("elapsed=%v got=%d want=%d", tc.elapsed, got, tc.want)
		}
	}
}

func TestDualVolgaCarrierIndexUsesSafeDefault(t *testing.T) {
	if got := dualVolgaCarrierIndex(25*time.Second, 0); got != 1 {
		t.Fatalf("default rotation got=%d want=1", got)
	}
}

func TestDualVolgaRotationDoesNotAgeWhileIdle(t *testing.T) {
	// The transport can stay warm for minutes before the first payload. Rotation
	// is intentionally based on the first Send() timestamp, so an idle warm-up
	// period is not part of carrier age. The index helper therefore sees zero
	// elapsed at first DATA and must begin on A.
	if got := dualVolgaCarrierIndex(0, 25*time.Second); got != 0 {
		t.Fatalf("first DATA must start on A, got=%d", got)
	}
}
